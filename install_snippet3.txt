363:     local config_file=$1
364:     local command=$2
365: 
366:     if grep -Fxq "$command" "$config_file"; then
367:         print_message info "Command already exists in $config_file, skipping write."

371:         print_message info "${MUTED}Successfully added ${NC}opencode ${MUTED}to \$PATH in ${NC}$config_file"
372:     else
373:         print_message warning "Manually add the directory to $config_file (or similar):"

435:                 print_message info "  export PATH=$INSTALL_DIR:\$PATH"
436:             ;;
437:         esac
438:     fi
